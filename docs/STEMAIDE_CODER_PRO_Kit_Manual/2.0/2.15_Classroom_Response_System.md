## Mounting the component on the breadboard and wiring the circuit

**Step 1:**	Place the two push buttons and the I2C LCD display on the breadboard. Ensure that the push buttons are positioned correctly across the center gap of the breadboard.

- Connect one terminal of the first push button to Digital Pin 2 on the Arduino Uno using a jumper wire. Connect the other terminal of the push button to GND using a jumper wire.

- Connect one terminal of the second push button to Digital Pin 3 on the Arduino Uno using a jumper wire. Connect the other terminal of the push button to GND using a jumper wire.

- Connect the VCC pin of the I2C LCD display to the 5V pin on the Arduino Uno. Then, connect the GND pin of the LCD display to GND using jumper wires.

- Connect the SDA pin of the I2C LCD display to A4 on the Arduino Uno. Then, connect the SCL pin to A5 using jumper wires.

- Ensure that the GND connections of both push buttons and the I2C LCD display are connected to the same ground line. A common ground is required for the components to operate correctly.

## PROGRAMMING

**Step 1:** Open your Arduino IDE. See how to set up here: [Getting Started](../../Getting Started/Arduino_IDE_Setup.md).

**Step 2:** Write the complete program implementing the system logic with appropriate pin definitions, setup configuration, and the main control loop.

```cpp

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

// Button Pins
const int buttonA = 2;
const int buttonB = 3;
const int buttonC = 4;
const int buttonD = 5;

// Vote Counters
int votesA = 0;
int votesB = 0;
int votesC = 0;
int votesD = 0;

// Previous button states
bool lastA = HIGH;
bool lastB = HIGH;
bool lastC = HIGH;
bool lastD = HIGH;

void setup()
{
  Serial.begin(9600);

  pinMode(buttonA, INPUT_PULLUP);
  pinMode(buttonB, INPUT_PULLUP);
  pinMode(buttonC, INPUT_PULLUP);
  pinMode(buttonD, INPUT_PULLUP);

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0,0);
  lcd.print("Class Response");
  lcd.setCursor(0,1);
  lcd.print("System Ready");

  delay(2000);

  updateDisplay();
}

void loop()
{
  checkButtons();
}

void checkButtons()
{
  bool currentA = digitalRead(buttonA);
  bool currentB = digitalRead(buttonB);
  bool currentC = digitalRead(buttonC);
  bool currentD = digitalRead(buttonD);

  if(lastA == HIGH && currentA == LOW)
  {
    votesA++;
    showVote("Vote A");
  }

  if(lastB == HIGH && currentB == LOW)
  {
    votesB++;
    showVote("Vote B");
  }

  if(lastC == HIGH && currentC == LOW)
  {
    votesC++;
    showVote("Vote C");
  }

  if(lastD == HIGH && currentD == LOW)
  {
    votesD++;
    showVote("Vote D");
  }

  lastA = currentA;
  lastB = currentB;
  lastC = currentC;
  lastD = currentD;
}

void showVote(String option)
{
  Serial.println(option);

  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print(option);
  lcd.setCursor(0,1);
  lcd.print("Recorded");

  delay(800);

  updateDisplay();
}

void updateDisplay()
{
  lcd.clear();

  lcd.setCursor(0,0);
  lcd.print("A:");
  lcd.print(votesA);
  lcd.print(" B:");
  lcd.print(votesB);

  lcd.setCursor(0,1);
  lcd.print("C:");
  lcd.print(votesC);
  lcd.print(" D:");
  lcd.print(votesD);
}

```
