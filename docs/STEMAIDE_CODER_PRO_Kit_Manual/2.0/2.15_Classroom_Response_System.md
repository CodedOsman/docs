## Mounting the component on the breadboard and wiring the circuit

**Step 1:**	Place the Push Buttons and LCD Display on the breadboard.

**Step 2:** Place the Ultrasonic Sensor and the Servo Motor on the breadboard or in suitable positions for wiring.

**Step 3:** Connect the Ultrasonic Sensor by connecting VCC to 5V, GND to GND, Trig to Digital Pin 9, and Echo to Digital Pin 10.

**Step 4:** Connect the Servo Motor by connecting VCC to 5V, GND to GND, and the Signal wire to Digital Pin 6.

**Step 5:** Since both the Ultrasonic Sensor and Servo Motor require 5V and GND, first connect the Arduino 5V pin to the breadboard's positive (+) power rail and the Arduino GND pin to the negative (-) power rail. Then connect the VCC and GND pins of both components to the corresponding breadboard power rails. Ensure that all GND connections share the same ground line to maintain stable and reliable circuit operation.

**Step 6:** Check the polarity of the servo motor and ultrasonic sensor, and verify that all wiring connections are correct before connecting the Arduino Uno to your computer via the USB cable.

## PROGRAMMING

**Step 1:** Open your Arduino IDE. See how to set up here: [Getting Started](../../Getting Started/Arduino_IDE_Setup.md).

**Step 2:** Write the complete program implementing the system logic with appropriate pin definitions, setup configuration, and the main control loop.

```cpp

/*
==========================================================
STEMAIDE Africa
Project 30 - Classroom Response System
Board: Arduino Uno
==========================================================

Button A -> D2
Button B -> D3
Button C -> D4
Button D -> D5

LCD SDA -> A4
LCD SCL -> A5
LCD VCC -> 5V
LCD GND -> GND
==========================================================
*/

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

// Button Pins
const int buttonA = 2;
const int buttonB = 3;
const int buttonC = 4;
const int buttonD = 5;

// Vote Counters
int votesA = 0;
int votesB = 0;
int votesC = 0;
int votesD = 0;

// Previous button states
bool lastA = HIGH;
bool lastB = HIGH;
bool lastC = HIGH;
bool lastD = HIGH;

void setup()
{
  Serial.begin(9600);

  pinMode(buttonA, INPUT_PULLUP);
  pinMode(buttonB, INPUT_PULLUP);
  pinMode(buttonC, INPUT_PULLUP);
  pinMode(buttonD, INPUT_PULLUP);

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0,0);
  lcd.print("Class Response");
  lcd.setCursor(0,1);
  lcd.print("System Ready");

  delay(2000);

  updateDisplay();
}

void loop()
{
  checkButtons();
}

void checkButtons()
{
  bool currentA = digitalRead(buttonA);
  bool currentB = digitalRead(buttonB);
  bool currentC = digitalRead(buttonC);
  bool currentD = digitalRead(buttonD);

  if(lastA == HIGH && currentA == LOW)
  {
    votesA++;
    showVote("Vote A");
  }

  if(lastB == HIGH && currentB == LOW)
  {
    votesB++;
    showVote("Vote B");
  }

  if(lastC == HIGH && currentC == LOW)
  {
    votesC++;
    showVote("Vote C");
  }

  if(lastD == HIGH && currentD == LOW)
  {
    votesD++;
    showVote("Vote D");
  }

  lastA = currentA;
  lastB = currentB;
  lastC = currentC;
  lastD = currentD;
}

void showVote(String option)
{
  Serial.println(option);

  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print(option);
  lcd.setCursor(0,1);
  lcd.print("Recorded");

  delay(800);

  updateDisplay();
}

void updateDisplay()
{
  lcd.clear();

  lcd.setCursor(0,0);
  lcd.print("A:");
  lcd.print(votesA);
  lcd.print(" B:");
  lcd.print(votesB);

  lcd.setCursor(0,1);
  lcd.print("C:");
  lcd.print(votesC);
  lcd.print(" D:");
  lcd.print(votesD);
}
```
