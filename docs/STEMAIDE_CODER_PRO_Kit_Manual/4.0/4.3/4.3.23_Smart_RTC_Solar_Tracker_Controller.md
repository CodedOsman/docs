## Mounting the component on the breadboard and wiring the circuit

**Step 1:** Inventory all modules listed in the Required Components table.

**Step 2:** Connect the DS3231 RTC (DATA to A4, CLK to A5, VCC to 5V, GND to GND) and the IIC 1602 LCD (SDA to A4, SCL to A5, VCC to 5V, GND to GND) to the Arduino's I2C bus.

**Step 3:** Connect the Photoresistor output to Analog Pin A0 and the Servo Motor signal wire to Digital Pin 9.

**Step 4:** Ensure all VCC and GND pins are correctly connected to the 5V and GND rails on the Arduino.

**Step 5:** Double-check all VCC, 5V, 3.3V, and GND polarities to prevent electrical shorts.

**Step 6:** Connect USB programming cable only after verifying all signal path wiring.

#include <Wire.h>
#include <RTClib.h>
#include <LiquidCrystal_I2C.h>
#include <Servo.h>

// Initialize components
// Ensure DS3231 DATA -> A4, CLK -> A5
RTC_DS3231 rtc;
LiquidCrystal_I2C lcd(0x27, 16, 2);
Servo myServo;

void setup() {
  Serial.begin(9600);
  Wire.begin();

  if (!rtc.begin()) {
    Serial.println("Couldn't find RTC");
    while (1);
  }

  // Check if RTC is running
  if (rtc.lostPower()) {
    Serial.println("RTC lost power, let's set the time!");
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  }

  lcd.init();
  lcd.backlight();
  myServo.attach(9);
  Serial.println("Project 143: Smart RTC Solar Tracker Controller Online");
}

void loop() {
  int lightValue = analogRead(A0);
  DateTime now = rtc.now();

  lcd.setCursor(0, 0);
  lcd.print("Time: ");
  lcd.print(now.hour());
  lcd.print(":");
  lcd.print(now.minute());

  // Simple tracker logic
  if (lightValue < 500) {
    myServo.write(90);
  } else {
    myServo.write(180);
  }

  delay(1000);
}
