# 4.1.15. Precision Climate Chamber Controller

| **Description** | In Project 70, learners will build a precision climate chamber monitor that uses a DS18B20 to control a DC motor through a relay when the chamber reaches the set temperature. The BME280 and LCD show ambient and chamber readings. |

| **Use case**     |This project can be applied to real-world IoT, safety, and control systems while developing essential master-level engineering skills.|

## Building the circuit

> **Tip:** Use the [ESP32 Pin Mapping](../../assets/esp32_pin_mapping.png) image to locate the correct GPIO pins on your ESP32 board.


Things Needed:

- ESP32 board Core Board = 1

- BME280 = 1

- DS18B20 = 1

- IIC 1602 LCD = 1

- USB Cable & Jumper Wires = 1

- Bread Board

- Bidirectional I2C logic-level shifter = 1 (if the LCD module requires 5V)

- Relay Module

- DC Motor = 1

- External Power Supply(9V Battery) = 1


## Wiring the circuit

**Step 1:** Connect the BME280 to its I2C bus:

- Connect the BME280 **SDA** to **GPIO 21**.
- Connect the BME280 **SCL** to **GPIO 22**.
- Connect the BME280 **GND** to common **GND**.
- Connect the BME280 **VCC** to **3.3V**.

![Circuit wiring step 1](../../../assets/project-files/expert/4.1.15_Precision_Climate_Chamber_Controller/circuit_1.png)

**Step 2:** Connect the I2C LCD display to a separate I2C bus:

- Connect the LCD **SDA** to **GPIO 32**.
- Connect the LCD **SCL** to **GPIO 33**.
- Connect the LCD **GND** to common **GND**.
- Connect the LCD **VCC** to **3.3V** if supported; otherwise connect it to **5V** and route the LCD **SDA** and **SCL** through a bidirectional I2C logic-level shifter.

![Circuit wiring step 2](../../../assets/project-files/expert/4.1.15_Precision_Climate_Chamber_Controller/circuit_2.png)

**Step 3:** Connect the DS18B20 temperature sensor:

- Connect **VDD** to **3.3V**
- Connect **GND** to common **GND**
- Connect **DQ** to **GPIO 4**.

![Circuit wiring step 3](../../../assets/project-files/expert/4.1.15_Precision_Climate_Chamber_Controller/circuit_3.png)

**Step 4:** Connect the relay-switched DC motor:

- Connect relay **IN** to **GPIO 26**
- Connect **VCC** to its rated supply
- Connect **GND** to common **GND**.
- Connect relay **COM** to the positive terminal of the motor-rated DC power supply.
- Connect relay **NO** to one DC motor terminal.
- Connect the other DC motor terminal to the power supply negative terminal.
- Use a DC motor and supply within the relay module's contact voltage and current ratings. The relay switches the motor on or off; it does not control motor speed.

![Circuit wiring step 4](../../../assets/project-files/expert/4.1.15_Precision_Climate_Chamber_Controller/circuit_4.png)



## PROGRAMMING

**Step 1:** Open your Arduino IDE. See how to set up here: [Getting Started](../../Getting Started/Arduino_IDE_Setup.md).

**Step 2:** Install the **Adafruit BME280**, **Adafruit Unified Sensor**, **OneWire**, and **DallasTemperature** libraries from Arduino IDE Library Manager, then enter and upload this complete program.

```cpp
#include <Wire.h>
#include <Adafruit_BME280.h>
#include <OneWire.h>
#include <DallasTemperature.h>

// Project 70: Precision Climate Chamber Controller
const int oneWireBus = 4;
const int relayPin = 26;
const int lcdSdaPin = 32;
const int lcdSclPin = 33;
const float motorOnTemperatureC = 30.0;
const bool relayActiveLow = false;
const uint8_t lcdAddress = 0x27;

const uint8_t lcdRegisterSelect = 0x01;
const uint8_t lcdEnable = 0x04;
const uint8_t lcdBacklight = 0x08;

Adafruit_BME280 bme;
TwoWire lcdI2c(1);
OneWire oneWire(oneWireBus);
DallasTemperature ds18b20(&oneWire);

void lcdWriteExpander(uint8_t value) {
  lcdI2c.beginTransmission(lcdAddress);
  lcdI2c.write(value | lcdBacklight);
  lcdI2c.endTransmission();
}

void lcdPulseEnable(uint8_t value) {
  lcdWriteExpander(value | lcdEnable);
  delayMicroseconds(1);
  lcdWriteExpander(value & ~lcdEnable);
  delayMicroseconds(50);
}

void lcdWriteNibble(uint8_t value, bool isData) {
  const uint8_t output = (value & 0xF0) | (isData ? lcdRegisterSelect : 0);
  lcdWriteExpander(output);
  lcdPulseEnable(output);
}

void lcdSend(uint8_t value, bool isData) {
  lcdWriteNibble(value, isData);
  lcdWriteNibble(value << 4, isData);
}

void lcdCommand(uint8_t command) {
  lcdSend(command, false);
  if (command == 0x01 || command == 0x02) delay(2);
}

void lcdInitialize() {
  delay(50);
  lcdWriteNibble(0x30, false);
  delay(5);
  lcdWriteNibble(0x30, false);
  delayMicroseconds(150);
  lcdWriteNibble(0x30, false);
  lcdWriteNibble(0x20, false);
  lcdCommand(0x28);
  lcdCommand(0x0C);
  lcdCommand(0x06);
  lcdCommand(0x01);
}

void lcdSetCursor(uint8_t column, uint8_t row) {
  const uint8_t rowOffsets[] = {0x00, 0x40};
  lcdCommand(0x80 | (column + rowOffsets[row % 2]));
}

void lcdPrint(const char *text) {
  while (*text) lcdSend(static_cast<uint8_t>(*text++), true);
}

void lcdPrintTemperature(float temperatureC) {
  char text[8];
  dtostrf(temperatureC, 4, 1, text);
  lcdPrint(text);
  lcdSend(0xDF, true);
  lcdPrint("C ");
}

void setup() {
  Serial.begin(9600);
  // BME280 uses I2C controller 0; the LCD uses controller 1 on separate pins.
  Wire.begin(21, 22);
  lcdI2c.begin(lcdSdaPin, lcdSclPin, 100000);
  lcdInitialize();
  pinMode(relayPin, OUTPUT);
  digitalWrite(relayPin, relayActiveLow ? HIGH : LOW);

  if (!bme.begin(0x76) && !bme.begin(0x77)) {
    Serial.println("BME280 not found at 0x76 or 0x77.");
    while (true) delay(100);
  }

  ds18b20.begin();
  Serial.println("Project 70: Precision Climate Chamber Controller Initialized...");
}

void loop() {
  ds18b20.requestTemperatures();
  const float chamberTemp = ds18b20.getTempCByIndex(0);
  const float ambientTemp = bme.readTemperature();
  const bool sensorValid = chamberTemp != DEVICE_DISCONNECTED_C;
  const bool motorShouldRun = sensorValid && chamberTemp >= motorOnTemperatureC;

  const bool relayOutputHigh = motorShouldRun != relayActiveLow;
  digitalWrite(relayPin, relayOutputHigh ? HIGH : LOW);

  lcdSetCursor(0, 0);
  lcdPrint("Chamber:");
  if (sensorValid) {
    lcdPrintTemperature(chamberTemp);
  } else {
    lcdPrint("Sensor error   ");
  }
  lcdSetCursor(0, 1);
  lcdPrint("Ambient:");
  lcdPrintTemperature(ambientTemp);

  Serial.print("Chamber: ");
  if (sensorValid) Serial.print(chamberTemp, 1);
  else Serial.print("sensor error");
  Serial.print(" C, ambient: ");
  Serial.print(ambientTemp, 1);
  Serial.print(" C, DC motor: ");
  Serial.println(motorShouldRun ? "ON" : "OFF");
  delay(1000);
}
```
